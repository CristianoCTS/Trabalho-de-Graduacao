% ==== Lê dados de cada ESP ====
dados = NaN(numESPs, 5);

for i = 1:numESPs
    try
        d = thingSpeakRead(espChIds(i),'Fields', 1:5, 'NumPoints', 1, 'ReadKey', espReadKeys{i});
        dados(i,:) = d;
    catch
        fprintf('Falha ao ler canal %d\n', espChIds(i));
    end
end
ctrl = thingSpeakRead(espChIds(3), 'Fields', 1:3,'NumPoints', 1, 'ReadKey', espReadKeys{3});
ControlType = ctrl(1);
SetPoint = ctrl(2);
GastoEnergetico = ctrl(3);
Ocupacao = max(dados(:,4), [], 'omitnan');

% ==== Garante 15 segundos desde o último write no broker ====
try
    [~, lastTime] = thingSpeakRead(writeChId, 'Field', 1, 'NumPoints', 1, 'ReadKey', readKeyBrk);
catch
    lastTime = [];
end
if isempty(lastTime)
    elapsed = minInterval;
else
    if isempty(lastTime.TimeZone)
        lastTime.TimeZone = 'UTC';
    end
    elapsed = seconds(datetime('now','TimeZone','UTC') - lastTime);
end
if elapsed < minInterval
    pause(minInterval - elapsed);
end

comand = BrokerControl(ControlType, SetPoint, Ocupacao)
codigo = comand
parametros = '020026023030020026100015030300027025020120100030020003004000';
FLAG = -1;
ctrlAnterior = GetLastControlType(writeChId, readKeyBrk);
houveTroca = isnan(ctrlAnterior) || (ctrlAnterior ~= ControlType);

energia = NaN(1,3);
ctValido = ismember(ControlType, 1:3);
NovoGasto = ~isnan(GastoEnergetico);

if houveTroca
    energia(:) = FLAG;
end

if ctValido && ~isnan(GastoEnergetico)
    energia(ControlType) = GastoEnergetico;
end

if NovoGasto || houveTroca
    thingSpeakWrite(writeChId, {codigo, parametros, energia(1), energia(2), energia(3)}, 'Fields', 1:5, 'WriteKey', writeKey);
else
     thingSpeakWrite(writeChId, {codigo, parametros}, 'Fields', [1 2], 'WriteKey', writeKey);
end

%==================================Funcoes================================================%

function comando = BrokerControl(ControlType, SetPoint, Ocupacao)
    if ControlType == 3 && Ocupacao <= 0
        msg1 = 9;
        msg2 = 9;
        ESPactive = 0;
    else
        msg1 = 1;
        msg2 = 1;
        ESPactive = 1;
    end
 
    comando = (SetPoint * 1e4) + (ControlType * 1e3) + (msg1 * 1e2) + (msg2 * 1e1) + ESPactive;
end
function ct = GetLastControlType(writeChId, readKeyBrk)
try
 valor = thingSpeakRead(writeChId, 'Field', 1, 'NumPoints', 1, 'ReadKey', readKeyBrk);
 if isnan(valor)
     ct = NaN;
 else
     ct = mod(floor(round(valor)/1e3), 10);
 end
catch
 ct = NaN;
end
end

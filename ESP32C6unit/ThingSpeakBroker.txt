%==================================Funcoes================================================%

function comando = BrokerControl(ControlType, SetPoint, Ocupacao)
    if ControlType == 3 && Ocupacao <= 0
        msg1 = 9;
        msg2 = 9;
        ESPactive = 0;
    else
        msg1 = 1;
        msg2 = 1;
        ESPactive = 1;
    end
 
    comando = (SetPoint * 1e4) + (ControlType * 1e3) + (msg1 * 1e2) + (msg2 * 1e1) + ESPactive;
end
function ct = GetLastControlType(writeChId, readKeyBrk)
    try
     valor = thingSpeakRead(writeChId, 'Field', 1, 'NumPoints', 1, 'ReadKey', readKeyBrk);
        if isnan(valor)
            ct = NaN;
        else
            ct = mod(floor(round(valor)/1e3), 10);
        end
    catch
     ct = NaN;
    end
end
function ocp = GetOcupation(a, b, c, d, o, h)
    da = floor(a/1e4);  
    ta = mod(a, 1e4);
    db = floor(b/1e4);  
    tb = mod(b, 1e4);
    try
        if (c == 2)
            if (tb >= h)
                ocp = o + db;
            else
                ocp = o + da;
            end
        elseif (d == 2)
            if (ta >= h)
                ocp = o + da;
            else
                ocp = o + db;
            end
        elseif c == 8 || d == 8
            ocp = 0;
        elseif abs(ta - tb) > 5
            ocp = o + (da + db);
        else
            ocp = o + max(da, db);
         end
    catch
     ocp = o;
    end
    ocp = max(ocp, 0);
end
function alrt = Alertas(a, b)
    alrt1 = 0;
    alrt2 = 0;
    try
        switch a
            case 0
                %pass%
            case 1
                %pass%
            case 2
                %pass%
            case 3
                %pass%
            case 4
                %pass%
            case 5
                %pass%
            case 6
                alrt1 = 6;
            case 7
                alrt1 = 7;
            case 8
                %pass%
            case 9
                %pass%
            otherwise
                disp('Modo inválido')
        end
        switch b
            case 0
                %pass%
            case 1
                %pass%
            case 2
                %pass%
            case 3
                %pass%
            case 4
                %pass%
            case 5
                %pass%
            case 6
                alrt2 = 6;
            case 7
                alrt2 = 7;
            case 8
                %pass%
            case 9
                %pass%
            otherwise
                disp('Modo inválido')
        end
    catch
        alrt1 = 0;
        alrt2 = 0;
    end
    alrt = (alrt1 * 10) + alrt2;
end

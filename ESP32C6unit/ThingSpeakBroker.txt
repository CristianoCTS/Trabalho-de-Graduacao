%===============================Funcoes====================================%

function comando = BrokerControl(ControlType, SetPoint, Ocupacao)
    if (((ControlType == 3) || (ControlType == 5)) && Ocupacao <= 0)
        msg1 = 9;
        msg2 = 9;
        ESPactive = 0;
    else
        msg1 = 1;
        msg2 = 1;
        ESPactive = 1;
    end
 
    comando = (SetPoint * 1e4) + (ControlType * 1e3) + (msg1 * 1e2) + (msg2 * 1e1) + ESPactive;
end
function IntervaloMinimo(minInterval, writeChId, readKeyBrk)
    try
        [~, lastTime] = thingSpeakRead(writeChId, 'Field', 1, 'NumPoints', 1, 'ReadKey', readKeyBrk);
    catch
        lastTime = [];
    end
    if isempty(lastTime)
        elapsed = minInterval;
    else
        if isempty(lastTime.TimeZone)
            lastTime.TimeZone = 'UTC';
        end
        elapsed = seconds(datetime('now','TimeZone','UTC') - lastTime);
    end
    if elapsed < minInterval
        pause(minInterval - elapsed);
    end
end
function ct = GetLastControlType(writeChId, readKeyBrk)
    try
     valor = thingSpeakRead(writeChId, 'Field', 1, 'NumPoints', 1, 'ReadKey', readKeyBrk);
        if isnan(valor)
            ct = NaN;
        else
            ct = mod(floor(round(valor)/1e3), 10);
        end
    catch
     ct = NaN;
    end
end
function ocp = GetOcupation(a, b, c, d, o)
    if isnan(a), a = 0; end
    if isnan(b), b = 0; end
    try
        if c == 8 || d == 8
            ocp = 0;
        else
            ocp = o + a + b;
        end
    catch
        ocp = o;
    end
    ocp = max(ocp, 0);
end
function alrt = Alertas(a, b)
    alrt1 = 0;
    alrt2 = 0;
    try
        switch a
            case 0

            case 1
                alrt1 = 1;
            case 2
                alrt1 = 2;
            case 3
                alrt1 = 3;
            case 4
                alrt1 = 4;
            case 5

            case 6
                alrt1 = 6;
            case 7
                alrt1 = 7;
            case 8
                alrt1 = 8;
            case 9
                alrt1 = 9;
            otherwise
                disp('Modo inválido')
        end
        switch b
            case 0
                alrt2 = 0;
            case 1
                alrt2 = 1;
            case 2

            case 3
                alrt2 = 3;
            case 4
                alrt2 = 4;
            case 5
                alrt2 = 5;
            case 6
                alrt2 = 6;
            case 7
                alrt2 = 7;
            case 8
                alrt2 = 8;
            case 9
                alrt2 = 9;
            otherwise
                disp('Modo inválido')
        end
    catch
        alrt1 = 0;
        alrt2 = 0;
    end
    alrt = (alrt1 * 10) + alrt2;
end
